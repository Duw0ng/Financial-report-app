function buildMonthlySummary(){const months={};for(const t of state.transactions){if(t.internalTransfer)continue;const k=monthKey(t.date);months[k]??={Mes:k,'Ingresos ARS':0,'Gastos ARS':0,'Balance ARS':0,'Ingresos USD':0,'Gastos USD':0,'Balance USD':0};const r=months[k],key=`${t.type==='credit'?'Ingresos':'Gastos'} ${t.currency}`;r[key]+=t.amount;r[`Balance ${t.currency}`]+=t.type==='credit'?t.amount:-t.amount}return Object.values(months).sort((a,b)=>a.Mes.localeCompare(b.Mes))}
